-- STREAMING_CHUNK:Fixing PostgreSQL function syntax & RLS permissions in schema...
-- =========================================================================
-- CODENIX ID SECURE GATEWAY - CORRECTED POSTGRESQL / SUPABASE DATABASE SCHEMA
-- =========================================================================

-- Enable UUID extension for secure identifier generation
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. PROFILES TABLE (Linked with Supabase Auth Users)
-- STREAMING_CHUNK:Creating profiles table...
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email TEXT UNIQUE NOT NULL,
    display_name TEXT,
    virtual_id TEXT UNIQUE,
    avatar_url TEXT,
    bio TEXT,
    privacy_accepted BOOLEAN DEFAULT FALSE,
    mfa_completed BOOLEAN DEFAULT FALSE,
    role TEXT DEFAULT 'user',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 2. SIGNED-IN USER DEVICES & SESSIONS TABLE
CREATE TABLE IF NOT EXISTS public.user_devices (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    device_name TEXT NOT NULL,
    browser TEXT NOT NULL,
    os TEXT NOT NULL,
    ip_address TEXT,
    location TEXT,
    is_current BOOLEAN DEFAULT false,
    last_active TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. ECOSYSTEM APPLICATIONS TABLE
CREATE TABLE IF NOT EXISTS public.ecosystem_apps (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    app_slug TEXT UNIQUE NOT NULL,
    app_name TEXT NOT NULL,
    description TEXT,
    icon_class TEXT DEFAULT 'fa-cubes',
    category TEXT DEFAULT 'Developer Tool',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. LINKED ECOSYSTEM USER ACCOUNTS & TOKENS TABLE
CREATE TABLE IF NOT EXISTS public.user_linked_apps (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    app_slug TEXT NOT NULL REFERENCES public.ecosystem_apps(app_slug) ON DELETE CASCADE,
    granted_scopes TEXT[] DEFAULT ARRAY['Read Profile', 'Fast Add Auth'],
    linked_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    last_used TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id, app_slug)
);

-- STREAMING_CHUNK:Enabling Row Level Security...
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_devices ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_linked_apps ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.ecosystem_apps ENABLE ROW LEVEL SECURITY;

-- 5. SECURE RLS POLICIES FOR AUTHENTICATED USERS
-- STREAMING_CHUNK:Configuring RLS policies...
DROP POLICY IF EXISTS "Public profiles are viewable by everyone." ON public.profiles;
CREATE POLICY "Public profiles are viewable by everyone." 
    ON public.profiles FOR SELECT 
    USING (true);

DROP POLICY IF EXISTS "Users can insert their own profile." ON public.profiles;
CREATE POLICY "Users can insert their own profile." 
    ON public.profiles FOR INSERT 
    TO authenticated
    WITH CHECK (auth.uid() = id);

DROP POLICY IF EXISTS "Users can update own profile." ON public.profiles;
CREATE POLICY "Users can update own profile." 
    ON public.profiles FOR UPDATE 
    TO authenticated
    USING (auth.uid() = id)
    WITH CHECK (auth.uid() = id);

DROP POLICY IF EXISTS "Users can view own devices" ON public.user_devices;
CREATE POLICY "Users can view own devices" ON public.user_devices FOR SELECT USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Users can delete own devices" ON public.user_devices;
CREATE POLICY "Users can delete own devices" ON public.user_devices FOR DELETE USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Users can insert own devices" ON public.user_devices;
CREATE POLICY "Users can insert own devices" ON public.user_devices FOR INSERT WITH CHECK (auth.uid() = user_id);

DROP POLICY IF EXISTS "Users can view linked apps" ON public.user_linked_apps;
CREATE POLICY "Users can view linked apps" ON public.user_linked_apps FOR SELECT USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Users can manage linked apps" ON public.user_linked_apps;
CREATE POLICY "Users can manage linked apps" ON public.user_linked_apps FOR ALL USING (auth.uid() = user_id);

DROP POLICY IF EXISTS "Anyone can read apps catalog" ON public.ecosystem_apps;
CREATE POLICY "Anyone can read apps catalog" ON public.ecosystem_apps FOR SELECT USING (true);

-- 6. AUTO-CREATE PROFILE TRIGGER FUNCTION (CORRECTED)
-- STREAMING_CHUNK:Creating auto-profile trigger function...
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER SET search_path = ''
AS $$
BEGIN
    INSERT INTO public.profiles (id, email, display_name, virtual_id, privacy_accepted, mfa_completed)
    VALUES (
        NEW.id, 
        NEW.email, 
        COALESCE(NEW.raw_user_meta_data->>'display_name', split_part(NEW.email, '@', 1)),
        COALESCE(NEW.raw_user_meta_data->>'virtual_id', split_part(NEW.email, '@', 1)),
        COALESCE((NEW.raw_user_meta_data->>'privacy_accepted')::boolean, FALSE), 
        FALSE
    )
    ON CONFLICT (id) DO UPDATE SET
        email = EXCLUDED.email,
        updated_at = NOW();
    RETURN NEW;
END;
$$;

-- 7. ATTACH TRIGGER TO AUTH.USERS
-- STREAMING_CHUNK:Attaching trigger to auth.users...
DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
    AFTER INSERT ON auth.users
    FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();

-- 8. FUNCTION: RECORD DEVICE ON SIGN IN
CREATE OR REPLACE FUNCTION public.record_user_device(
    p_device_name TEXT,
    p_browser TEXT,
    p_os TEXT,
    p_ip TEXT,
    p_location TEXT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO public.user_devices (user_id, device_name, browser, os, ip_address, location, last_active, is_current)
    VALUES (auth.uid(), p_device_name, p_browser, p_os, p_ip, p_location, now(), true);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 9. FUNCTION: COMPLETE USER ACCOUNT DELETION
CREATE OR REPLACE FUNCTION public.delete_user_account()
RETURNS VOID AS $$
DECLARE
    v_user_id UUID := auth.uid();
BEGIN
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Not authenticated';
    END IF;

    -- Delete user profile & related data (Cascades via Foreign Keys)
    DELETE FROM public.profiles WHERE id = v_user_id;
    DELETE FROM auth.users WHERE id = v_user_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
