Supabase SQL Tables (Prefix: nixy_)

Run this SQL in your Supabase SQL Editor to create all required nixy_* tables:

-- 1. User Quotas, IP Tracking & Codenix Tokens
create table if not exists nixy_user_limits (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null unique,
  email text,
  ip_address text,
  minute_window timestamptz default now(),
  minute_count int default 0,
  day_date date default current_date,
  daily_chat_count int default 0,
  daily_image_count int default 0,
  codenix_tokens_used bigint default 0,
  puter_linked boolean default false,
  updated_at timestamptz default now()
);

-- 2. Saved Conversations
create table if not exists nixy_conversations (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null,
  title text not null default 'New Conversation',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- 3. Saved Messages
create table if not exists nixy_messages (
  id uuid primary key default gen_random_uuid(),
  conversation_id uuid references nixy_conversations(id) on delete cascade,
  user_id uuid not null,
  role text not null,
  content text not null,
  tool_used text,
  provider_used text,
  tokens_used int default 0,
  created_at timestamptz default now()
);

-- 4. PIN-Protected Published Conversations
create table if not exists nixy_shared_chats (
  id uuid primary key default gen_random_uuid(),
  slug text unique not null,
  conversation_id uuid references nixy_conversations(id) on delete cascade,
  owner_id uuid not null,
  pin_code text not null,
  title text,
  messages_snapshot jsonb not null,
  created_at timestamptz default now()
);

-- 5. HostNix Published Apps (from Cody)
create table if not exists nixy_hostnix_apps (
  id uuid primary key default gen_random_uuid(),
  slug text unique not null,
  owner_id uuid not null,
  html_content text not null,
  created_at timestamptz default now()
);
