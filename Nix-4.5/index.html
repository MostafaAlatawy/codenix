Supabase SQL Schema Script

Run this query in your Supabase SQL editor to create message history tables:

CREATE TABLE nix_conversations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id),
  title TEXT NOT NULL,
  model TEXT DEFAULT 'claude-3-5-sonnet',
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);
CREATE TABLE nix_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  conversation_id UUID REFERENCES nix_conversations(id) ON DELETE CASCADE,
  role TEXT NOT NULL,
  content TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);
